import Base

# SOURCE DEVELOPMENT ONLY. Payload types are arbitrary affine Type.
# Schema constructors define kind once; product structure derives lenses.
type Component<-S:Data,-P:Type> is Data:
  Component{key:String}
type Resource<-S:Data,-P:Type> is Data:
  Resource{key:String}
type RegistryKind is Data:
  ComponentKind{}
  ResourceKind{}
type Entry<-S:Data> is Data:
  Entry{kind:RegistryKind,id:U32,key:String,name:String}
type Pair<-S:Data,A:Data,B:Data> is Data:
  Pair{left:A,right:B}

def component(~S:Data,~P:Type,key:String) -> Component<S,P>:
  Component{key}
def resource(~S:Data,~P:Type,key:String) -> Resource<S,P>:
  Resource{key}
def pair(~S:Data,~A:Data,~B:Data,left:A,right:B) -> Pair<S,A,B>:
  Pair{left,right}
# Kind cannot be supplied separately. Name defaults to the sole declared key.
def component_entry(~S:Data,~P:Type,value:Component<S,P>,id:U32) -> Entry<S>:
  match value:
    case Component{+key}:Entry{ComponentKind{},id,key,key}
def resource_entry(~S:Data,~P:Type,value:Resource<S,P>,id:U32) -> Entry<S>:
  match value:
    case Resource{+key}:Entry{ResourceKind{},id,key,key}
# Standard product lens, generated by the library; no author storage observer.
def take_left(~L:Type,~R:Type,store:L & R) -> R & L:
  match store:
    case (left,right):(right,left)
def put_left(~L:Type,~R:Type,right:R,left:L) -> L & R:
  (left,right)
def take_right(~L:Type,~R:Type,store:L & R) -> L & R:
  store
def put_right(~L:Type,~R:Type,left:L,right:R) -> L & R:
  (left,right)
