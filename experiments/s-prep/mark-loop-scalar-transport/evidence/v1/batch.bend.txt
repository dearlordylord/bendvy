import Base
import ./measurement-bend.bend as M
import /tmp/bendvy-mark-scalar-native/experiments/s-integrate/types.bend as T
import /tmp/bendvy-mark-scalar-native/experiments/s-integrate/storage.bend as S

import /tmp/bendvy-mark-scalar-native/experiments/s-integrate/schedule.bend as K

def motion_prepared(fuel:Nat,+count:U32,owners:List<(D.Runtime<M.MotionBench,S.Handle<T.MotionSchema>> & K.System)>) -> IO(List<(D.Runtime<M.MotionBench,S.Handle<T.MotionSchema>> & K.System)>):
  match fuel:
    case 0n: IO.pure(List<(D.Runtime<M.MotionBench,S.Handle<T.MotionSchema>> & K.System)>,List.reverse(&1,(D.Runtime<M.MotionBench,S.Handle<T.MotionSchema>> & K.System),owners))
    case 1n+rest: IO.bind((D.Runtime<M.MotionBench,S.Handle<T.MotionSchema>> & K.System),List<(D.Runtime<M.MotionBench,S.Handle<T.MotionSchema>> & K.System)>,M.motion_fresh(count),owner => motion_prepared(rest,count,owner <> owners))
def motion_returned(next:List<D.Runtime<M.MotionBench,S.Handle<T.MotionSchema>>> -> IO(List<D.Runtime<M.MotionBench,S.Handle<T.MotionSchema>>>),done:List<D.Runtime<M.MotionBench,S.Handle<T.MotionSchema>>>,owner:D.Runtime<M.MotionBench,S.Handle<T.MotionSchema>>) -> IO(List<D.Runtime<M.MotionBench,S.Handle<T.MotionSchema>>>):
  next(owner <> done)
def motion_execute(owners:List<(D.Runtime<M.MotionBench,S.Handle<T.MotionSchema>> & K.System)>,done:List<D.Runtime<M.MotionBench,S.Handle<T.MotionSchema>>>) -> IO(List<D.Runtime<M.MotionBench,S.Handle<T.MotionSchema>>>):
  match owners:
    case Nil{}: IO.pure(List<D.Runtime<M.MotionBench,S.Handle<T.MotionSchema>>>,List.reverse(&1,D.Runtime<M.MotionBench,S.Handle<T.MotionSchema>>,done))
    case Con{Tuple{owner,system},rest}: IO.bind(D.Runtime<M.MotionBench,S.Handle<T.MotionSchema>>,List<D.Runtime<M.MotionBench,S.Handle<T.MotionSchema>>>,M.motion_loops(64n,system,owner),value => motion_returned(values => motion_execute(rest,values),done,value))
def motion_dump(owners:List<D.Runtime<M.MotionBench,S.Handle<T.MotionSchema>>>) -> IO(Unit):
  match owners:
    case Nil{}: IO.pure(Unit,Unit{})
    case Con{owner,rest}:
      do IO<Unit>:
        M.motion_dump(0n,owner)
        motion_dump(rest)
def motion_timed(owners:List<(D.Runtime<M.MotionBench,S.Handle<T.MotionSchema>> & K.System)>) -> IO(Unit):
  do IO<Unit>:
    start:Nat <- IO.now()
    returned:List<D.Runtime<M.MotionBench,S.Handle<T.MotionSchema>>> <- motion_execute(owners,[])
    end:Nat <- IO.now()
    IO.print("BATCH-MILLISECONDS:" ++ Nat.show(Nat.sub(end,start)))
    motion_dump(returned)
def motion_batch(+count:U32,batch:U32) -> IO(Unit):
  do IO<Unit>:
    M.motion_start(count,False{})
    prepared:List<(D.Runtime<M.MotionBench,S.Handle<T.MotionSchema>> & K.System)> <- motion_prepared(U32.to_nat(batch),count,[])
    motion_timed(prepared)
