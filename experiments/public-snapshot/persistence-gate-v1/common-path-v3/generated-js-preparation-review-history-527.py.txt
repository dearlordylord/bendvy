def encode(value):
    if isinstance(value,bytes):return {'base64Bytes':base64.b64encode(value).decode()}
    if isinstance(value,dict):return {k:encode(v) for k,v in value.items()}
    if isinstance(value,list):return [encode(v) for v in value]
    return value

def decode(value):
    if isinstance(value,dict):
        if set(value)=={'base64Bytes'}:return base64.b64decode(value['base64Bytes'],validate=True)
        return {k:decode(v) for k,v in value.items()}
    if isinstance(value,list):return [decode(v) for v in value]
    return value

def execute(record,logs,label,argv,cap,env,cwd):
    attempt={'label':label,'argv':list(argv),'capSeconds':cap,'attemptState':'ATTEMPTED'}
    record['commands'].append(attempt)
    try:result=task_runner.execute_result(argv,cap,env,cwd,'split')
    except BaseException as error:
        attempt['exception']=f'{type(error).__name__}: {error}';raise
    attempt.update({k:v for k,v in result.items() if not isinstance(v,bytes)})
    attempt['attemptState']='RETURNED'
    attempt['streams']={k:{'sha256':__import__('hashlib').sha256(result[k]).hexdigest(),'bytes':len(result[k])} for k in ('stdout','stderr')}
    logs.record(label,result['stdout'],result['stderr'])
    record['logs']=dict(logs.hashes)
    return result
