def cells(view:P.View) -> List<&2,U32>:
  match view:
    case P.View{first,second}: [first,second]
def physical_payload(next:Array<Maybe<&1,P.Payload>> -> List<&2,U32>,values:Array<Maybe<&1,P.Payload>>,result:P.Payload & P.View) -> List<&2,U32>:
  match result:
    case (_,view): List.append(&2,U32,cells(view),next(values))
def physical_taken(next:Array<Maybe<&1,P.Payload>> -> List<&2,U32>,result:Array<Maybe<&1,P.Payload>> & Maybe<&1,P.Payload>) -> List<&2,U32>:
  match result:
    case (values,None{}): List.append(&2,U32,[999,999],next(values))
    case (values,Some{owner}): physical_payload(next,values,P.project(owner))
def physical(fuel:Nat,+id:U32,values:Array<Maybe<&1,P.Payload>>) -> List<&2,U32>:
  match fuel:
    case 0n: []
    case 1n+rest: physical_taken((values => physical(rest,U32.add(id,1),values)),Array.swap(Maybe<&1,P.Payload>,values,U32.sub(id,1),None{}))
def recovered(+points:List<&2,U32>,column:Col.Column<P.Schema,P.Payload>) -> List<&2,U32>:
  match column:
    case Col.Column{values,_}: List.append(&2,U32,points,physical(4n,1,values))
    case _: []
def projected(+points:List<&2,U32>,column:Col.Column<P.Schema,P.Payload>,result:P.Payload & P.View) -> List<&2,U32>:
  match result:
    case (_,view): recovered(List.append(&2,U32,points,cells(view)),Col.recover(~P.Schema,~P.Payload,column))
def returned(+status:U32,column:Col.Column<P.Schema,P.Payload>,owner:Maybe<&1,P.Payload>) -> List<&2,U32>:
  match owner:
    case None{}: recovered([status,999,999],Col.recover(~P.Schema,~P.Payload,column))
    case Some{owner}: projected([status],column,P.project(owner))
def observed(result:Col.SwapOutcome<P.Schema,P.Payload>) -> List<&2,U32>:
  match result:
    case Col.Accepted{column,owner}: returned(0,column,owner)
    case Col.Rejected{column,owner}: returned(1,column,owner)
def rows() -> H.PrototypeHandoffRows<P.Schema,P.Payload>:
  H.HandoffCon{1,P.Payload{ANode{ALeaf{3},ALeaf{7}}},H.HandoffCon{3,P.Payload{ANode{ALeaf{11},ALeaf{19}}},H.HandoffNil{}}}
def probe(legacy:Bool,+fuel:Nat,+target:U32) -> List<&2,U32>:
  match legacy:
    case True{}: observed(Col.seek(~P.Schema,~P.Payload,fuel,Col.absent(~P.Payload,2n),[],4,target,Some{P.Payload{ANode{ALeaf{41},ALeaf{43}}}},rows(),Col.HistoryNil{},Col.Inspect{}))
    case False{}: observed(Col.seek_owned(~P.Schema,~P.Payload,fuel,Col.absent(~P.Payload,2n),[],4,target,Some{P.Payload{ANode{ALeaf{41},ALeaf{43}}}},Col.HistoryNil{},Col.Ready{rows()}))
def compared_probe(legacy:Bool,+fuel:Nat,+equal:Bool,+before:Bool) -> List<&2,U32>:
  match legacy:
    case True{}: observed(Col.seek(~P.Schema,~P.Payload,fuel,Col.absent(~P.Payload,2n),[],4,2,Some{P.Payload{ANode{ALeaf{41},ALeaf{43}}}},rows(),Col.HistoryNil{},Col.Choice{equal,before}))
    case False{}: observed(Col.seek_owned(~P.Schema,~P.Payload,fuel,Col.absent(~P.Payload,2n),[],4,2,Some{P.Payload{ANode{ALeaf{41},ALeaf{43}}}},Col.HistoryNil{},Col.Compared{1,P.Payload{ANode{ALeaf{3},ALeaf{7}}},H.HandoffCon{3,P.Payload{ANode{ALeaf{11},ALeaf{19}}},H.HandoffNil{}},equal,before}))
