# Non-failing body only. Operation refusals are owned outputs, never silently
# converted into system failure or thrown away. Fallible transactional assembly
# remains an explicit integration boundary.

    case Args{target,input}:completed(~S,~C,~R,~E,~P,~O,~identity,~project,~admit,~U,Public.invoke(~S,~(W.World<S,C,R,E> & W.Handle<S>),~P,~O,~A,~U,~identity,~project,~admit,~body,Decl.constructed(~S,~C,~R,~E,~P,~O,~identity,~project,~admit,declaration),(world,target),input))

    case Sys.RegistrationFailed{world}:Ordinary.Rejected{world}
